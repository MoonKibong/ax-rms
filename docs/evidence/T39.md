### 기준 12 (g) 확인 스크립트

저장소에 두지 않은 셸 확인이라 전문을 남긴다(저장소 루트에서 `bash check-12g.sh <로그 폴더>`). 레드를 기록한 뒤 바꾼 곳은 sentinel이
없어졌을 때의 `cksum` 줄 하나다: 레드 실행에서는 `.nexi-tmp/e2e-write.db: No such file or directory` 오류 줄이 출력에 섞였으므로
`$( (cksum < "$db") 2>/dev/null || echo missing)`로 오류 출력만 감췄다. 단언은 그대로다.

```bash
#!/usr/bin/env bash
# T#39 criterion 12 (g), review REVISE reason 1: a second scripts/e2e-servers.sh started while another run is still
# starting (building, both ports still free) is refused before it removes, writes or starts anything, and the first
# run finishes on its own servers and DB. Pairs: shared → --fresh-db, --fresh-db → --fresh-db, --fresh-db → shared.
# usage (from the repository root): bash check-12g.sh <folder for the logs>
set -u
work=$1
db=.nexi-tmp/e2e-write.db
shared_before=$(sha256sum backend/data/e2e.db | cut -d' ' -f1)
failed=0

pair() { # <name> <first run's option or ""> <second run's option or "">
  local name=$1 a_mode=$2 b_mode=$3 sentinel_sum="" sum_after_b="" a_pid a_status b_status verdict=PASS
  if [ -z "$a_mode" ]; then
    printf 'sentinel %s\n' "$name" > "$db"
    sentinel_sum=$(cksum < "$db")
  fi
  bash scripts/e2e-servers.sh $a_mode -- bash -c '
    sleep 20
    curl -sf -o /dev/null http://localhost:14173/api/health || { echo "A: own servers gone"; exit 1; }
    if [ "${E2E_DB:-}" = fresh ] && [ ! -f .nexi-tmp/e2e-write.db ]; then echo "A: own DB removed"; exit 1; fi
    echo "A: ok"' > "$work/$name.a.log" 2>&1 &
  a_pid=$!
  sleep 1
  bash scripts/e2e-servers.sh $b_mode -- bash -c 'echo "B: ran (E2E_DB=${E2E_DB:-none})"' > "$work/$name.b.log" 2>&1
  b_status=$?
  [ -z "$sentinel_sum" ] || sum_after_b=$( (cksum < "$db") 2>/dev/null || echo missing)
  wait "$a_pid"
  a_status=$?

  [ "$b_status" -ne 0 ] && grep -q '다른 e2e-servers 실행이 진행 중' "$work/$name.b.log" || verdict=FAIL
  ! grep -q '^B: ran' "$work/$name.b.log" || verdict=FAIL
  [ -z "$sentinel_sum" ] || [ "$sum_after_b" = "$sentinel_sum" ] || verdict=FAIL
  [ "$a_status" -eq 0 ] && grep -q '^A: ok' "$work/$name.a.log" || verdict=FAIL
  echo "$name: second exit=$b_status, refused=$(grep -c '다른 e2e-servers 실행이 진행 중' "$work/$name.b.log")," \
    "second ran=$(grep -c '^B: ran' "$work/$name.b.log")," \
    "sentinel=${sentinel_sum:+$([ "$sum_after_b" = "$sentinel_sum" ] && echo same || echo "changed ($sum_after_b)")}," \
    "first exit=$a_status ($(grep -E '^A: ' "$work/$name.a.log" | tail -n 1)) → $verdict"
  [ "$verdict" = PASS ] || failed=1
  rm -f "$db" # only this check's sentinel; a --fresh-db run has removed its own files already
}

pair shared-then-fresh "" --fresh-db
pair fresh-then-fresh --fresh-db --fresh-db
pair fresh-then-shared --fresh-db ""

left=$(ls .nexi-tmp/e2e-write.db* 2>/dev/null | wc -l)
shared_after=$(sha256sum backend/data/e2e.db | cut -d' ' -f1)
ports=$(for p in 18080 14173; do curl -s -o /dev/null -m 2 "http://localhost:$p/" && echo "$p"; done | tr '\n' ' ')
echo "after: DB files left=$left, shared DB $([ "$shared_before" = "$shared_after" ] && echo same || echo changed)" \
  "($shared_after), ports answering=[${ports% }]"
[ "$left" -eq 0 ] && [ "$shared_before" = "$shared_after" ] && [ -z "$ports" ] || failed=1
echo "exit=$failed"
exit "$failed"
```
